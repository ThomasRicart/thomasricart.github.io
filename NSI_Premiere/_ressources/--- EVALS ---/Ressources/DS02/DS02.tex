\begin{center}\begin{large}
DS02A - Structures IF / FOR / WHILE
\end{large}\end{center}



\begin{enumerate}
	\item 
	
	\begin{lstlisting}[numbers=none]
def f(a,b):
	if a < b:
		return a
	else:
		return b
\end{lstlisting}
	
	Que renvoie la commande \verb+f(3,7)+ ? \dotfill
	
	\medskip
	
	\item 
	\begin{lstlisting}[numbers=none]
def f(x):
	for i in range(6):
		x = x + i
	return x
\end{lstlisting}
	
	Que renvoie la commande \verb+f(1)+ ? \dotfill

	\medskip
	
	\item 
	\begin{lstlisting}[numbers=none]
mon_tuple = (1,6,8,3,2)
def f(un_tuple:tuple):
	a = 0
	for elt in un_tuple:
		a = a + elt
	return a
\end{lstlisting}
	
	Quelle est la valeur renvoyée par \verb+f(mon_tuple)+? \dotfill

	\medskip
	
	\item 
	\begin{lstlisting}[numbers=none]
def f(x):
	i = 0
	a = 0
	while i < x:
		a = a + 3
		i = i + 1
	return a
\end{lstlisting}
	
	Quelle est la valeur renvoyée par \verb+f(3)+? \dotfill
	
	\medskip
	
	\item 
	\begin{lstlisting}[numbers=none]
def f(n):
	for i in range(n+1):
		if i % 2 == 0:
			print(i , end= " ")
\end{lstlisting}
	
	Quelle est la valeur renvoyée par \verb+f(9)+? \dotfill
	
	
	\newpage

\begin{lstlisting}[numbers=none]

		Entrée:

\end{enumerate}
