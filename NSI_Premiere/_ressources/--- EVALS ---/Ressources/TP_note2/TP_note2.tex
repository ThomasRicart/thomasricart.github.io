On considère l'algorithme de dichotomie suivant:

\begin{lstlisting}
def recherche_dichotomique(L:list,valeur:int):
	''' Algorithme de recherche dichotomique '''
    i_deb = 0
    i_fin = len(L) - 1
    while i_fin - i_deb >= 0:
        i_cen = (i_deb + i_fin) // 2
        if L[i_cen] == valeur:
            return True
        elif valeur > L[i_cen]:
            i_deb = i_cen + 1
        else:
            i_fin = i_cen - 1
    return False
\end{lstlisting}
