\begin{definition}[Programmation Orientée Objet - POO]{}
    La \textbf{Programmation Orientée Objet} ou \textbf{POO} est un paradigme de programmation informatique. Il consiste en la définition et l'interaction de briques logicielles appellées \verb+objets+. 

    Chaque objet est une \textbf{instance} de \textbf{classes} qui définissent le comportement et les propriétés des objets.
\end{definition}

\medskip

\begin{exemple}[Exemple de classes et d'objets]{\quad}{}
    \begin{itemize}
        \item \verb+Pokemon+ est une classe; c'est un concept. Pour le moment aucun pokemon n'existe mais juste l'idée;
        \item \verb+pikachu+ est un objet. C'est un pokemon particulier. On dit que c'est une instance de classe;
        \item une classe possède un nom et des caractéristiques (on parle d'\textbf{attributs});
        \item une classe possède des fonctionnalités (on parle de \textbf{méthodes})
        
    \end{itemize}
\end{exemple}

    \begin{propriete}{En python tout est objet}{}
    En python, toutes les variables sont déjà des objets. Pour vous en convaincre: \verb+dir(list)+

    Dans la liste des méthodes associées à la classe \verb+list+, on retrouve les méthodes connues (\verb+append+, \verb+pop+...) mais également des méthodes spéciales (ou \textbf{méthodes magiques}) qui sont entourées de double underscores (\verb+__init__+, \verb+__len__+, \verb+__add__+, etc.). Ces méthodes permettent de définir le comportement des objets de la classe.
\end{propriete}


\section{Construire une classe}

\begin{definition}[Construire une classe]{}
    Une classe est au minimum constituée d'un constructeur de classe.

    \begin{CodePythontexAlt}[Largeur=1\linewidth,Centre,Lignes=true,PremLigne=1]{}
        class Point:                            # Définition du nom de la classe
            def __init__(self, abs, ord):       # Constructeur de classe
                self.x = abs                    # Attribut x
                self.y = ord                    # Attribut y
    \end{CodePythontexAlt}
\end{definition}

    \begin{itemize}
        \item \verb+class Point+: c'est la porte d'entrée de la classe;
        \item \verb+def __init__(self, abs, ord)+: c'est la méthode qui est appelée au moment de la création d'une instance de la classe \verb+Point+;
        \item \verb+abs+ et \verb+ord+ sont des paramètre à entrer au moment de créer l'instance;
        \item \verb+self+: c'est l'objet en lui même. Pour le moment, il n'a pas de nom puisqu'on crée la classe. Ensuite \verb+self+ sera remplacé par le nom de l'instance de la classe;
        \item \verb+self.x+: on crée un attribut de classe \verb+x+ qui prend l'abscisse pour valeur;
        \item \verb+self.y+: on crée un attribut de classe \verb+y+ qui prend l'ordonnée pour valeur.

    \end{itemize}

    
\begin{definition}[Se servir d'une classe]{}
    
    Pour se servir d'une classe, il suffit maintenant de créer un objet (une instance) de la classe \verb+Point+

    \begin{CodePythontexAlt}[Largeur=1\linewidth,Centre,Lignes=true,PremLigne=1]{}
    A = Point(1, 7)     # A est une instance de la classe Point
    B = Point(0, 4)     # B est également une instance de la classe Point
\end{CodePythontexAlt}

Au moment de la création de l'instance A, le constructeur est exécuté. Dans toute la suite, \verb+self+ sera donc remplacé par \verb+A+.


\begin{CodePythontexAlt}[Largeur=1\linewidth,Centre,Lignes=true,PremLigne=1]{}
    print(A.x)     # On récupère l'attribut x de l'instance A. Affichage 1
    print(A.y)     # On récupère l'attribut y de l'instance A. Affichage 7

    dx = B.x - A.x    # On calcule la différence d'abscisse entre les points A et B
    dy = B.y - A.y    # On calcule la différence d'ordonnée entre les points A et B

    coeff_directeur = dy/dx   # On calcule le coefficient directeur de la droite (AB)
\end{CodePythontexAlt}

\end{definition}


\begin{definition}[Accesseurs et mutateurs]{}
En général on s'interdit d'accéder aux attributs de l'instance de la classe directement. A la place on crée des méthodes permettant d'accéder à ces attributs:

\begin{itemize}
    \item \textbf{Accesseurs} (getters): permet de récupérer la valeur de l'attribut;
    \item \textbf{Mutateurs} (setters): permet de modifier la valeur de l'attribut.
\end{itemize}

\begin{CodePythontexAlt}[Largeur=1\linewidth,Centre,Lignes=true,PremLigne=1]{}
    class Point:                                # Définition du nom de la classe
        def __init__(self, abs, ord):       # Constructeur de classe
            self.x = abs                    # Attribut x
            self.y = ord                    # Attribut y
        def get_coord(self):
            ''' Renvoie les coordonnées du point sous forme de tuple '''
            return (self.x, self.y)
        def set_coord(self, coord:tuple):
            ''' Modifie les coordonnées du point '''
            self.x, self.y = coord

    \end{CodePythontexAlt}

\end{definition}

\begin{easybox}[Affichage]{}
    Python permet d'afficher de deux manières: soit via la fonctionnalité \verb+print+ soit directement dans la console. Derrière ces fonctionnalités, il se cache deux méthodes:
    \begin{itemize}
        \item Pour la fonction \verb+print+, c'est la méthode \verb+__str__+
        \item Pour l'affichage dans la console c'est la méthode \verb+__repr__+
    \end{itemize}

    Par défaut ces méthodes sont crées lors de la création de la classe \verb+Point+ mais ne donnent rien d'intéressant:
        \begin{CodePythontexAlt}[Largeur=1\linewidth,Centre,Lignes=true,PremLigne=1]{}
        >>> print(A)
        <__main__.Point object at 0x00000233824A4D08>   
        # A est un objet de type Point situé à l'adresse mémoire 0x00000233824A4D08                 
    \end{CodePythontexAlt}

        On va donc surcharger (modifier) ces deux méthodes

        \begin{CodePythontexAlt}[Largeur=1\linewidth,Centre,Lignes=true,PremLigne=1]{}
            def __str__(self):
                ''' Surcharge la fonction print. Affichage 'Coordonnées = (abs;ord)'''
                return f'Coordonnées = ({self.x};{self.y})'
            def __repr__(self)
                ''' Surcharge __repr__. Affichage (abs;ord)'''
                return f'({self.x};{self.y})
        \end{CodePythontexAlt}
        


\end{easybox}

\begin{definition}[D'autres méthodes intéressantes]{}
    Dans le même ordre d'idée, il est possible de redéfinir des méthodes comme:
    \begin{itemize}
        \item \verb+__eq__(self, objet)+ qui renvoie \verb+True+ si \verb+self == objet+;
        \item \verb+__gt__(self, objet)+ qui renvoie \verb+True+ si \verb+self > objet+;
        \item \verb+__lt__(self, objet)+ qui renvoie \verb+True+ si \verb+self < objet+;
        \item \verb+__add__(self, objet)+ qui renvoie \verb-self + objet-;
        \item \verb+__mul__(self, objet)+ qui renvoie \verb+self * objet+;
    \end{itemize}
\end{definition}

\begin{propriete}[Utilisation des méthodes]{}

    \begin{CodePythontexAlt}[Largeur=1\linewidth,Centre,Lignes=true,PremLigne=1]{}
        >>> A = Point(1, 6)
        >>> B = Point(2, 7)
        >>> C = Point(1, 6)
        >>> B.set_coord((4, 8))
        >>> B.get_coord()
        (4, 8)
        >>> print(B)
        Coordonnées = (4;8)
        >>> B
        (4;8)
        >>> A == B
        False
        >>> A == C
        True
    \end{CodePythontexAlt}
    
\end{propriete}

\begin{easybox}{Le nom des variables en Python}
	\begin{itemize}
		\item Elles doivent commencer par une lettre ou un underscore (\texttt{\_}).

\begin{easybox}{Les chaines de caractères}{}
    \smallskip
    Les chaines de caractères sont des variables qui contiennent du texte. Elles sont délimitées par des guillemets simples ou doubles.

    \begin{minipage}{0.32\linewidth}

            b = ' a tous'

            print(c)
        \end{ConsolePythontex}
    \end{minipage}\hfill
    \begin{minipage}{0.32\linewidth}
        \begin{ConsolePythontex}[ Label=false]{}

        \end{ConsolePythontex}
    \end{minipage}\hfill
    \begin{minipage}{0.32\linewidth}
        \begin{ConsolePythontex}[ Label=false]{}
            a = 'Bonjour'

        \end{ConsolePythontex}
    \end{minipage}

    \medskip

\begin{easybox}{Les opérations sur les chaines de caractères}{}   
    \smallskip
    \begin{itemize}
        \item[] \texttt{+} \quad pour la concaténation:  \verb~'Bonjour' + ' a tous'~ donne \verb~'Bonjour a tous'~

\begin{easybox}{Booléen comme résultat de comparaison}{}
    Les booléens sont le résultat d'une comparaison. Les opérateurs de comparaison sont:
    \smallskip
    \begin{itemize}
        \begin{minipage}{0.44\linewidth}
            \item[] \texttt{==} \quad pour le test d'égalité
            \item[] \texttt{!=} \quad pour le test de différence
            \item[] \texttt{<} \quad pour le test d'infériorité
        \end{minipage}
        \begin{minipage}{0.44\linewidth}
            \item[] \texttt{>} \quad pour le test de supériorité
            \item[] \texttt{<=} \quad pour le test d'infériorité ou d'égalité
            \item[] \texttt{>=} \quad pour le test de supériorité ou d'égalité
        \end{minipage}
    \end{itemize}

    \end{easybox}
